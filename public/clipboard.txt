export type ReadRequest = {
  select?: string[];
  where?: WhereNode;
  order_by?: OrderBy[];
  limit?: number;
  offset?: number;
};

export type WhereOperator =
  | "="
  | "<"
  | ">"
  | "<="
  | ">="
  | "<>"
  | "LIKE"
  | "NOT LIKE"
  | "ILIKE"
  | "NOT ILIKE"
  | "IN"
  | "NOT IN"
  | "BETWEEN"
  | "IS NULL"
  | "IS NOT NULL";

export type WhereLogic = "AND" | "OR";

export type WhereCondition = {
  column: string;

  operator: WhereOperator;

  value: unknown;
};

export type WhereGroup = {
  logic: WhereLogic;
  conditions: WhereNode[];
};

export type WhereNode = WhereCondition | WhereGroup;

export type OrderDirection = "ASC" | "DESC";

export interface OrderBy {
  column: string;
  direction: OrderDirection;
}

export interface ReadResponse {
  message: string;
  select: string[];
  where: WhereNode | null;
  order_by: OrderBy[] | null;
  limit: number | null;
  offset: number | null;
  rows_count: number;
  result: Record<string, unknown>[];
}

export type ResolvedReadRequest = {
  select: string[];
  where?: WhereNode;
  order_by: OrderBy[];
  limit?: number;
  offset?: number;
};
